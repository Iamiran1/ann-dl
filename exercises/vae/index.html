    4. VAE

    Enunciado

    Exercises → VAE

    O fio condutor da atividade é o cabo de guerra dentro da loss. Um VAE paga por duas coisas ao mesmo tempo: reconstruir a entrada e manter o espaço latente perto de um prior do qual se pode amostrar. Primeiro um VAE de verdade no MNIST, registrando os dois termos separadamente; depois um espaço latente 2-D que dá para ver.

    Regras técnicas (valem para toda a atividade):

    • Fixar as sementes: torch.manual_seed(42) antes de construir cada modelo, e o gerador do split dado no Exercise 1;
    • Todo gráfico deve ter título e rótulos nos eixos; todo scatter, legenda das classes;
    • Bibliotecas permitidas: torch e torchvision, numpy, matplotlib/seaborn, scipy/scikit-learn para utilidades. A reparametrização, os dois termos da loss e a amostragem devem ser escritos pelo aluno;
    • Reportar os dois termos da loss separadamente, em nats por imagem (somados nos 784 pixels, média sobre as imagens);
    • O conjunto de teste só é usado para os números finais de cada modelo;
    • Todo número pedido deve aparecer no texto, não só na saída do código.

def to_tensors(dataset):
    """Empilha um dataset de imagens (1, 28, 28) em X (N, 784) e y (N,)."""
    X = torch.stack([img for img, _ in dataset]).view(-1, 784)  # achata 28x28 -> 784
    y = torch.tensor([label for _, label in dataset])
    return X, y


X_train, y_train = to_tensors(train)
X_val, y_val = to_tensors(val)
X_test, y_test = to_tensors(test)  # só usado nos números finais

import torch.nn as nn
import torch.nn.functional as F

class VAE(nn.Module):
    def __init__(self, d):
        super().__init__()
        # encoder: 784 -> 512 -> 256, ReLU depois de cada
        self.encoder = nn.Sequential(
            nn.Linear(784, 512), nn.ReLU(),
            nn.Linear(512, 256), nn.ReLU(),
        )
        # duas cabeças lineares 256 -> d (sem ativação!)
        self.head_mu = nn.Linear(256, d)
        self.head_logvar = nn.Linear(256, d)
        # decoder: d -> 256 -> 512 -> 784, sigmoid na saída
        self.decoder = nn.Sequential(
            nn.Linear(d, 256), nn.ReLU(),
            nn.Linear(256, 512), nn.ReLU(),
            nn.Linear(512,784), nn.Sigmoid(),
        )
    def encode(self, x):
        h = self.encoder(x)
        mu = self.head_mu(h)
        logvar = self.head_logvar(h)
        return mu, logvar
    def reparameterize(self,mu,logvar ):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        z = mu + std * eps
        return z

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        x_hat = self.decoder(z)
        return x_hat, mu, logvar


def vae_loss(x_hat, x, mu, logvar):
    """Devolve os dois termos da loss separados, em nats por imagem."""
    batch = x.shape[0]
    # reconstrução: BCE somada nos 784 pixels e dividida pelo nº de imagens
    # (reduction="mean" dividiria também por 784 e encolheria esse termo)
    recon = F.binary_cross_entropy(x_hat, x, reduction="sum") / batch
    # KL na forma fechada: ½ Σ (μ² + σ² − 1 − log σ²), somada nas d dimensões
    kl = 0.5 * torch.sum(mu.pow(2) + logvar.exp() - 1 - logvar) / batch
    return recon, kl


@torch.no_grad()  # só avalia: não calcula gradiente nem mexe nos pesos
def evaluate(model, X, batch_size=1000):
    """Reconstrução e KL médias (nats por imagem) de um conjunto inteiro."""
    model.eval()
    recon_total, kl_total = 0.0, 0.0
    for i in range(0, len(X), batch_size):
        x = X[i:i + batch_size]
        x_hat, mu, logvar = model(x)
        recon, kl = vae_loss(x_hat, x, mu, logvar)
        recon_total += recon.item() * len(x)  # volta à soma do batch, para fazer a média no fim
        kl_total += kl.item() * len(x)
    return recon_total / len(X), kl_total / len(X)


def train_vae(d, epochs=20, batch_size=128, lr=1e-3):
    """Treina um VAE com latente de dimensão d e devolve o modelo e o histórico por época."""
    torch.manual_seed(42)  # semente antes de construir o modelo (regra técnica)
    model = VAE(d)
    opt = torch.optim.Adam(model.parameters(), lr=lr)
    hist = {"train_recon": [], "train_kl": [], "val_recon": [], "val_kl": []}

    for epoch in range(epochs):
        model.train()
        perm = torch.randperm(len(X_train))  # embaralha o treino a cada época
        recon_total, kl_total = 0.0, 0.0
        for i in range(0, len(X_train), batch_size):
            x = X_train[perm[i:i + batch_size]]
            x_hat, mu, logvar = model(x)              # forward
            recon, kl = vae_loss(x_hat, x, mu, logvar)
            loss = recon + kl                         # o que o otimizador minimiza
            opt.zero_grad()
            loss.backward()                           # gradiente de todos os pesos
            opt.step()                                # ajusta os pesos
            recon_total += recon.item() * len(x)
            kl_total += kl.item() * len(x)

        # treino: média da época; validação: avaliada no fim da época
        hist["train_recon"].append(recon_total / len(X_train))
        hist["train_kl"].append(kl_total / len(X_train))
        val_recon, val_kl = evaluate(model, X_val)
        hist["val_recon"].append(val_recon)
        hist["val_kl"].append(val_kl)
        print(f"d={d} época {epoch + 1:2d}: treino recon {hist['train_recon'][-1]:.2f} "
              f"KL {hist['train_kl'][-1]:.2f} | val recon {val_recon:.2f} KL {val_kl:.2f}")

    return model, hist

    A — Load and split the data

    Enunciado

    Use o MNIST do torchvision, com pixels em \([0, 1]\), achatados em vetores de 784. Divida as 60 000 imagens de treino em 50 000 para treino e 10 000 para validação com random_split(..., generator=torch.Generator().manual_seed(42)). As 10 000 imagens oficiais de teste são o conjunto de teste.

    Os dados são carregados com transforms.ToTensor(), que já deixa os pixels em \([0, 1]\), e o split é exatamente o do enunciado:

    train_full = datasets.MNIST("data", train=True, download=True, transform=transforms.ToTensor())
test = datasets.MNIST("data", train=False, download=True, transform=transforms.ToTensor())
train, val = random_split(train_full, [50_000, 10_000],
                          generator=torch.Generator().manual_seed(42))

    Cada imagem sai do torchvision com formato \((1, 28, 28)\). A função to_tensors empilha todas as imagens de cada conjunto em um único tensor e achata cada uma em um vetor de 784 pixels, guardando também os rótulos:

    Conjunto X y
    Treino \((50\,000,\ 784)\) \((50\,000)\)
    Validação \((10\,000,\ 784)\) \((10\,000)\)
    Teste \((10\,000,\ 784)\) \((10\,000)\)

    Empilhar uma única vez deixa o treino bem mais rápido na CPU do que achatar cada batch a cada época. O conjunto de teste é carregado aqui, mas só é usado depois do treino, para os números finais e para as Figuras 2 e 3.

    B — Build the model

    Enunciado

    Parte Camadas
    Encoder \(784 \to 512 \to 256\), ReLU depois de cada
    Heads duas lineares \(256 \to d\): uma para \(\mu\), outra para \(\log \sigma^2\)
    Decoder \(d \to 256 \to 512 \to 784\), ReLU nas ocultas, sigmoide na saída

    com dimensão latente \(d = 16\). A loss por imagem, somada nos pixels e nas dimensões latentes:

    \[ L = -\sum_{j=1}^{784}\Big[x_j \log \hat{x}_j + (1-x_j)\log(1-\hat{x}_j)\Big] + \frac{1}{2}\sum_{i=1}^{d}\Big(\mu_i^2 + \sigma_i^2 - 1 - \log\sigma_i^2\Big) \]

    com \(\mathbf{z} = \mu + \sigma \odot \varepsilon\), \(\varepsilon \sim \mathcal{N}(0, I)\), média sobre as imagens do batch.

    Arquitetura. A classe VAE recebe d como parâmetro, para ser reaproveitada no Exercise 2 com \(d = 2\). No __init__ são criadas as camadas; os métodos encode, reparameterize e forward ligam essas camadas:

    def encode(self, x):
    h = self.encoder(x)              # 784 -> 256
    mu = self.head_mu(h)             # 256 -> d
    logvar = self.head_logvar(h)     # 256 -> d
    return mu, logvar

def forward(self, x):
    mu, logvar = self.encode(x)
    z = self.reparameterize(mu, logvar)
    x_hat = self.decoder(z)          # d -> 784, cada pixel em (0, 1)
    return x_hat, mu, logvar

    As duas cabeças têm a mesma forma (\(256 \to d\)), mas pesos próprios. O que dá a cada uma o seu papel é onde a saída é usada: \(\mu\) entra somada em \(\mathbf{z}\), e \(\log\sigma^2\) entra multiplicando o ruído. A rede prevê \(\log\sigma^2\) em vez de \(\sigma\) porque o log pode assumir qualquer valor real, enquanto \(\sigma\) precisa ser positivo. Por isso as cabeças não têm ativação.

    Reparametrização. O \(\mathbf{z}\) é sorteado em torno de \(\mu\), mas o sorteio fica todo no \(\varepsilon\), que não depende de nenhum peso:

    def reparameterize(self, mu, logvar):
    std = torch.exp(0.5 * logvar)    # sigma = exp(½ log sigma²)
    eps = torch.randn_like(std)      # epsilon ~ N(0, I), mesmo formato de std
    z = mu + std * eps
    return z

    Assim, \(\mathbf{z}\) é uma conta comum com \(\mu\) e \(\sigma\), e o gradiente consegue passar por ela até o encoder.

    Loss. A função devolve os dois termos separados, em nats por imagem:

    def vae_loss(x_hat, x, mu, logvar):
    batch = x.shape[0]
    recon = F.binary_cross_entropy(x_hat, x, reduction="sum") / batch
    kl = 0.5 * torch.sum(mu.pow(2) + logvar.exp() - 1 - logvar) / batch
    return recon, kl
    • A reconstrução usa reduction="sum" e divide pelo tamanho do batch. Com reduction="mean", ela seria dividida também pelos 784 pixels e ficaria 784 vezes menor em relação à KL.
    • A KL é a forma fechada do enunciado escrita à mão: mu.pow(2) é \(\mu^2\), logvar.exp() é \(\sigma^2\) e logvar é \(\log\sigma^2\).

    Como verificação, com o modelo ainda sem treino e 4 imagens de treino, a reconstrução foi 544.72 nats e a KL 0.04 nats. A reconstrução é próxima de \(784 \cdot \ln 2 \approx 543\), o valor de um modelo que prevê 0.5 em todos os pixels, e a KL é pequena porque no início \(\mu \approx 0\) e \(\sigma \approx 1\).

    C — Train

    Enunciado

    Treine com Adam, taxa de aprendizado \(10^{-3}\), batch 128, por 20 épocas. Ao fim de cada época, registre a reconstrução e a KL no treino (média da época) e na validação. Reporte a reconstrução, a KL e a soma (o ELBO negativo) no teste, em nats por imagem. Produza a Figura 1: dois painéis, reconstrução por época e KL por época, cada um com as curvas de treino e de validação.

    A função train_vae(d) fixa torch.manual_seed(42), cria o modelo e o Adam e, em cada época:

    1. embaralha as imagens de treino;
    2. para cada batch de 128: forward → vae_loss → loss = recon + kl → backward() → step();
    3. guarda a média da época de reconstrução e KL no treino;
    4. calcula reconstrução e KL na validação com evaluate, que roda sem gradiente (@torch.no_grad()).

    import numpy as np
import torch

# reaproveita do Exercise 1: dados já achatados, a classe VAE, o treino e a avaliação
# (o ex1_vae.py só treina quando é rodado diretamente, então importar não retreina nada)
from ex1_vae import VAE, train_vae, evaluate, X_test, y_test, FIGURES

# ---------------------------------------------------------------------------
# A — VAE 2-D: mesmo modelo, dados, semente, β e orçamento; só muda d
# ---------------------------------------------------------------------------
model2, hist2 = train_vae(d=2)
# o modelo d = 16 do Exercise 1 é treinado de novo aqui: train_vae fixa
# torch.manual_seed(42) antes de tudo, então ele sai idêntico ao do ex1_vae.py
model16, hist16 = train_vae(d=16)

recon2, kl2 = evaluate(model2, X_test)
recon16, kl16 = evaluate(model16, X_test)
print(f"teste d=16: recon {recon16:.2f}  KL {kl16:.2f}  -ELBO {recon16 + kl16:.2f}")
print(f"teste d=2 : recon {recon2:.2f}  KL {kl2:.2f}  -ELBO {recon2 + kl2:.2f}")

model2.eval()
model16.eval()

# ---------------------------------------------------------------------------
# B — Figura 4: μ das 10 000 imagens de teste, colorido por dígito
# ---------------------------------------------------------------------------
with torch.no_grad():
    mu_test, _ = model2.encode(X_test)  # (10000, 2): usa μ, sem sortear z
mu_test = mu_test.numpy()
labels = y_test.numpy()

fig, ax = plt.subplots(figsize=(8, 7))
cmap = plt.get_cmap("tab10")
for digit in range(10):
    pts = mu_test[labels == digit]
    ax.scatter(pts[:, 0], pts[:, 1], s=3, alpha=0.6, color=cmap(digit), label=str(digit))
ax.set_title("Figura 4 — μ das 10 000 imagens de teste no espaço latente (d = 2)")
ax.set_xlabel("$\\mu_1$")
ax.set_ylabel("$\\mu_2$")
ax.legend(title="Dígito", markerscale=4, loc="center left", bbox_to_anchor=(1, 0.5))
fig.savefig(FIGURES / "fig04-latent-scatter.png", dpi=150, bbox_inches="tight")
plt.close(fig)

# centro de cada dígito no plano, para localizar as sobreposições no texto
for digit in range(10):
    c = mu_test[labels == digit].mean(axis=0)
    print(f"dígito {digit}: centro de μ = ({c[0]:+.2f}, {c[1]:+.2f})")

# ---------------------------------------------------------------------------
# C — Figura 5: plano decodificado numa grade 15×15 em quantis do prior
# ---------------------------------------------------------------------------
n = 15
q = np.linspace(0.05, 0.95, n)  # 15 probabilidades igualmente espaçadas
grid = norm.ppf(q)              # z = Φ⁻¹(q): inversa da CDF da N(0, 1)
tick_labels = [f"{abs(v) if abs(v) < 1e-9 else v:.2f}" for v in grid]  # evita "-0.00" no centro
print("valores de z na grade:", np.round(grid, 2))

canvas = np.zeros((n * 28, n * 28))
with torch.no_grad():
    for row, z2 in enumerate(grid[::-1]):  # z2 cresce de baixo para cima, como num gráfico
        for col, z1 in enumerate(grid):
            z = torch.tensor([[z1, z2]], dtype=torch.float32)
            img = model2.decoder(z).view(28, 28).numpy()
            canvas[row * 28:(row + 1) * 28, col * 28:(col + 1) * 28] = img

fig, ax = plt.subplots(figsize=(9, 9))
ax.imshow(canvas, cmap="gray")
centers = np.arange(n) * 28 + 14
ax.set_xticks(centers)
ax.set_xticklabels(tick_labels, rotation=90)
ax.set_yticks(centers)
ax.set_yticklabels(tick_labels[::-1])
ax.set_xlabel("$z_1 = \\Phi^{-1}(q)$")
ax.set_ylabel("$z_2 = \\Phi^{-1}(q)$")
ax.set_title("Figura 5 — Plano latente decodificado, grade 15×15 em quantis da N(0, 1) (d = 2)")
fig.savefig(FIGURES / "fig05-latent-plane.png", dpi=150, bbox_inches="tight")
plt.close(fig)

# ---------------------------------------------------------------------------
# D — Figura 6: interpolação em pixels × no latente (com o modelo d = 16)
# ---------------------------------------------------------------------------
IDX_A, IDX_B = 3, 2  # índices de teste escolhidos (dígitos diferentes)
x_a, x_b = X_test[IDX_A], X_test[IDX_B]
print(f"interpolação: índice {IDX_A} (dígito {y_test[IDX_A].item()}) -> "
      f"índice {IDX_B} (dígito {y_test[IDX_B].item()})")

ts = torch.linspace(0, 1, 10)
with torch.no_grad():
    pixel_row = [(1 - t) * x_a + t * x_b for t in ts]           # mistura direta dos pixels
    mu_ab, _ = model16.encode(torch.stack([x_a, x_b]))
    mu_a, mu_b = mu_ab[0], mu_ab[1]
    z_path = torch.stack([(1 - t) * mu_a + t * mu_b for t in ts])  # caminho entre os μ
    latent_row = model16.decoder(z_path)

fig, axes = plt.subplots(2, 10, figsize=(13, 2.9))
for col, t in enumerate(ts):
    axes[0, col].imshow(pixel_row[col].view(28, 28), cmap="gray")
    axes[1, col].imshow(latent_row[col].view(28, 28), cmap="gray")
    axes[0, col].set_title(f"t = {t:.2f}", fontsize=9)
    for row in range(2):
        axes[row, col].set_xticks([])
        axes[row, col].set_yticks([])
axes[0, 0].set_ylabel("Pixels")
axes[1, 0].set_ylabel("Latente\n(d = 16)")
fig.suptitle(f"Figura 6 — Interpolação entre as imagens de teste {IDX_A} e {IDX_B}: "
             "em pixels (cima) e no espaço latente (baixo)")
fig.savefig(FIGURES / "fig06-interpolation.png", dpi=150, bbox_inches="tight")
plt.close(fig)
